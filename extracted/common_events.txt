==============================================================================
COMMON EVENT 001: Druckanzeige   Trigger: Parallel   Condition switch: [0001: initialize]
==============================================================================
  Variable [0049: Druck [Exp]] = Variable [0029: Druck [Exp+7&Val]]
  Variable [0049: Druck [Exp]] /= 10
  Variable [0049: Druck [Exp]] -= 7
  Variable [0050: Druck [Val]] = Variable [0029: Druck [Exp+7&Val]]
  Variable [0050: Druck [Val]] %= 10

==============================================================================
COMMON EVENT 002: Druck Tank   Trigger: Parallel   Condition switch: [0001: initialize]
==============================================================================
  Call Common Event: [0003: Saugleistung]
  Call Common Event: [0004: Enddruck]
  If Variable [0026: Jumbo door] == 1
    If Variable [0029: Druck [Exp+7&Val]] < 101
      Variable [0029: Druck [Exp+7&Val]] -= Variable [0047: Saugleistung]
    Else
      If Variable [0027: Handräder] == 2
        Variable [0029: Druck [Exp+7&Val]] -= Variable [0047: Saugleistung]
      Else
    If Variable [0029: Druck [Exp+7&Val]] < Variable [0045: Enddruck [Exp+7&Val]]
      Variable [0029: Druck [Exp+7&Val]] = Variable [0045: Enddruck [Exp+7&Val]]
    Else
  Else
    Variable [0029: Druck [Exp+7&Val]] = 101
  If Variable [0029: Druck [Exp+7&Val]] == 1
    Switch [0003: 1E-7 erreicht] = ON
  Else
  If Switch [0002: Doktorand weg] == OFF
    If Variable [0029: Druck [Exp+7&Val]] < 101
      Text: Das Klopfen im Tank ignorierend pumpst du die 
          : Kammer ab.
      Text: Als die Pumpe anzieht, höhrst du einen erstickten 
          : Schrei.
      Text: In deiner Panik, schaffst du es nicht, die Kammer 
          : rechtzeitig zu belüften.
      Text: Als du die Tür endlich öffnest, blicken dich die toten 
          : Augen eines Doktoranden anklagend an...
      Game Over
    Else
      If Variable [0026: Jumbo door] == 1
        Play SE: '041-Knock02' (vol 100, pitch 120)
        Wait: 5 frame(s)
        Play SE: '041-Knock02' (vol 100, pitch 120)
        Wait: 5 frame(s)
        Play SE: '041-Knock02' (vol 100, pitch 120)
        Wait: 5 frame(s)
        Play SE: '041-Knock02' (vol 100, pitch 120)
      Else
  Else
  Variable [0018: Zeit] += 1
  Wait: 400 frame(s)

==============================================================================
COMMON EVENT 003: Saugleistung   Trigger: None
==============================================================================
  Call Common Event: [0005: #Turbos]
  Variable [0047: Saugleistung] = 0
  If Variable [0046: #Turbos] == 0
    Variable [0047: Saugleistung] += Variable [0048: #Vorpumpen]
  Else
    Variable [0047: Saugleistung] += Variable [0046: #Turbos]
  Variable [0047: Saugleistung] += Variable [0042: Kryos]

==============================================================================
COMMON EVENT 004: Enddruck   Trigger: None
==============================================================================
  If Variable [0024: Ventil V1] == 0
    Call Common Event: [0005: #Turbos]
    If Variable [0046: #Turbos] > 0
      Variable [0045: Enddruck [Exp+7&Val]] = 12
      If Variable [0042: Kryos] == 1
        Variable [0045: Enddruck [Exp+7&Val]] = 1
      Else
    Else
  Else
    Variable [0045: Enddruck [Exp+7&Val]] = 51
  If Variable [0023: Ventil Zwichenvakuum Pumpe] == 1
    If Variable [0022: Ventil Zwichenvakuum atm] == 0
      If Variable [0043: Zwischenvakuumpumpe] == 1
      Else
        Variable [0045: Enddruck [Exp+7&Val]] += 10
    Else
      Variable [0045: Enddruck [Exp+7&Val]] += 10
  Else
    Variable [0045: Enddruck [Exp+7&Val]] += 10
